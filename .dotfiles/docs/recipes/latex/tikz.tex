\tikzset{circumcircle/.style n args={3}{%
% a ! p ! b can be used to interpolate two values a and b. The value p is used to determin which point on the interpolated line to select
insert path={let    \p1=($(#1)!0.5!(#2)$), % here p1 is defined to be center (at 0.5) between parameter #1 and #2
                    \p2=($(#1)!0.5!(#3)$),
                    % point!partway!angle:point can be used as modifier.
                    % The resulting point is rotate by the given angle around the first point.
                    \p3=($(#1)!0.5!(#2)!1!-90:(#2)$),
                    \p4=($(#1)!0.5!(#3)!1!90:(#3)$),
                    \p5=(intersection of \p1--\p3 and \p2--\p4)
                    in },
at={(\p5)},
circle through= {(#1)},circle={0.1cm}
}}
\def\scalep{0.4}
\begin{document}
    \begin{tikzpicture}
        % clip a tikzpicture
        \clip (-10*\scalep,-0.5) rectangle (10*\scalep,100*\scalep*\scalep);

        % draw a function, in this case a parabola
        \draw[domain = -5*\scalep:10*\scalep, variable = \x] plot ({\x}, {\x*\x*\scalep*\scalep0.5});


        % no spaces in foreach, otherwise, they are included in the variables and can not be interpreted as a number
        % count is really neat to enumerat, from can also be larger
        \foreach \nodename [count=\x from 1] in {A,B,C,D,E,F,G,H,I}
        {
            \ifodd\x
                \draw[fill, color=red!50!red] (\x*\scalep, \x*\x*\scalep*\scalep*0.5) circle (\scalep*0.4cm) node (\nodename){};
            \else
                \draw[fill, color=blue!50!blue] ({\x*\scalep}, {\x * \x*\scalep*\scalep*0.5}) circle (\scalep*0.4cm) node (\nodename){};
            \fi
        }
        \node[circumcircle={B}{F}{H}, draw=red] {};


\begin{document}
    \begin{tikzpicture}[thick]

        \node[circle, draw] (B) at (2, 0) {B};

        \node[->, thick] (A) -- (B) node [midway, above] {5};
